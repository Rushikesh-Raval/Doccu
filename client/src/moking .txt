import React, { useState } from "react";
import "./App.css";

function App() {
  const [messages, setMessages] = useState([
    {
      from: "agent",
      text: "Hi Rushikesh 👋, how can I help with your finances today?",
    },
  ]);
  const [input, setInput] = useState("");

  const handleSend = () => {
    if (!input.trim()) return;

    const userMessage = { from: "user", text: input };
    const agentReply = {
      from: "agent",
      text: mockAgentReply(input),
    };

    setMessages([...messages, userMessage, agentReply]);
    setInput("");
  };

  const mockAgentReply = (text) => {
    if (text.toLowerCase().includes("can i afford buying clothes")) {
      return "If you skip eating out for 3 weeks, you can save ₹5,000 and afford it! 💸";
    } else if (text.toLowerCase().includes("how much did i spend")) {
      return "You spent ₹3,250 on food last week 🍔.";
    } else if (
      text.toLowerCase().includes("how can i save to get a new phone")
    ) {
      return "Cut off your excess cost like weekends and unnecessary buying stuffs, then you will save 12000 a month. And you can afford the bike in a year. This is max we can do";
    } else {
      return "Let me crunch some numbers... 📊";
    }
  };

  return (
    <div className="app">
      <h1>💰 MoneyMind</h1>
      <div className="chat-box">
        {messages.map((msg, idx) => (
          <div
            key={idx}
            className={`chat-message ${msg.from === "user" ? "user" : "agent"}`}
          >
            {msg.text}
          </div>
        ))}
      </div>

      <div className="input-box">
        <input
          type="text"
          placeholder="Ask something like 'Can I afford a new phone?'"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => e.key === "Enter" && handleSend()}
        />
        <button onClick={handleSend} className="send">Send</button>
      </div>
    </div>
  );
}

export default App;
